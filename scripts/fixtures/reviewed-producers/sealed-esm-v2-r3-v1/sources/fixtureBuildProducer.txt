/** Static CLI checkpoint provenance. No inferred HMR/watch generation and no
 * hand-authored pack can stand in for the actual opened project's output. */
import childProcess, { spawn } from 'node:child_process'
import { createHash, randomUUID } from 'node:crypto'
import { accessSync, constants, existsSync, lstatSync, mkdirSync, readFileSync, readdirSync, readlinkSync, realpathSync, statSync, writeFileSync } from 'node:fs'
import { syncBuiltinESMExports, createRequire } from 'node:module'
import { basename, delimiter, dirname, isAbsolute, join, relative, resolve, sep } from 'node:path'
import { fileURLToPath } from 'node:url'
import { threadId } from 'node:worker_threads'
import { archiveCompilerBuildBytes, COMPILER_BUILD_PROFILE, inspectCompilerBuildEvidence } from './conformance-compiler-build.mjs'
import { inspectCompilerUseEvidence } from './conformance-compiler-use.mjs'
import { inspectExecutionFixture } from './pack-execution-evidence.mjs'

const MODULE = fileURLToPath(import.meta.url)
const PRELOAD = fileURLToPath(new URL('./conformance-fixture-packer-preload.mjs', import.meta.url))
const REPOSITORY = resolve(dirname(MODULE), '..')
export const FIXTURE_BUILD_PROFILE = 'static-cli-v1'
const CLI = 'packages/weapp-vite/bin/weapp-vite.js'
const PACKER = 'tools/container-manifest/bin/container-pack.mjs'
const PACKER_MODULES = ['tools/container-manifest/bin/container-pack.mjs', ...['index', 'json-integrity', 'rendering', 'wxs-dependencies', 'wxs-compiler', 'modern-view'].map(name => `tools/container-manifest/src/${name}.mjs`)]
const hash = bytes => createHash('sha256').update(bytes).digest('hex')
const canonical = value => Array.isArray(value) ? `[${value.map(canonical).join(',')}]` : value && typeof value === 'object'
  ? `{${Object.keys(value).sort().map(key => `${JSON.stringify(key)}:${canonical(value[key])}`).join(',')}}` : JSON.stringify(value)
const equal = (a, b) => canonical(a) === canonical(b)
const ensure = (ok, message) => { if (!ok) throw new Error(message) }
const inside = (root, file) => file === root || file.startsWith(root + sep)
const slash = path => path.split(sep).join('/')
const runtimeFacts = value => Object.fromEntries(['recordedAt', 'installationId', 'projectPath', 'tool', 'system'].map(key => [key, value[key]]))
const fileIdentities = files => files.map(({ artifact, ...item }) => ({ ...item, ...(artifact ? { sha256: artifact.sha256, bytes: artifact.bytes } : {}) }))
// Exact non-executable support outputs of the pinned CLI. New names still fail
// closed, including adjacent declaration files in this same directory.
const GENERATED_SUPPORT_PATHS = new Set(['tsconfig.json', ...['tsconfig.app.json', 'tsconfig.node.json', 'tsconfig.server.json', 'tsconfig.shared.json', 'tsconfig.shared.empty.d.ts', 'uni-app.d.ts', 'typed-router.d.ts'].map(path => `.weapp-vite/${path}`)])
// The pinned helper waits 3s for SIGTERM and 1s for stdio drain. A numeric
// 128+SIGTERM exit is evidence only when this observer saw that live child's
// successful cleanup request within the same bound.
const COMPANION_CLEANUP_BOUND_MS = 4_000
const recorderOwners = new Set()
// Deliberately closed shapes, shared by the observer and independent reader.
// A platform selection is not permission for unrelated compiler options.
function staticBuildArguments(args, cliPath) {
  return Array.isArray(args) && args.every(arg => typeof arg === 'string') && args[0] === cliPath
    && args[1] === 'build' && !!args[2] && !args[2].startsWith('-')
    && (args.length === 3 || args.length === 5 && args[3] === '--platform' && args[4] === 'weapp')
}
function selectedNode(command, options, cwd) {
  ensure(options.shell === undefined || options.shell === false, 'Static compiler shell invocation is unsupported')
  ensure(options.argv0 === undefined, 'Static compiler argv0 alias is unsupported')
  const selected = process.execPath, physical = realpathSync(selected)
  if (isAbsolute(command)) {
    ensure(realpathSync(command) === physical, 'Static compiler executable differs from the selected Node')
    return { method: 'absolute', selected, physical, executable: physical }
  }
  ensure(command === 'node' && process.platform !== 'win32', 'Static compiler executable name or platform is unsupported')
  const path = (options.env ?? process.env).PATH
  ensure(typeof path === 'string' && !path.includes('\0'), 'Static compiler bare node requires an explicit PATH')
  for (const directory of path.split(delimiter)) {
    const file = resolve(cwd, directory, 'node')
    try {
      if (!statSync(file).isFile()) continue
      accessSync(file, constants.X_OK)
    } catch (error) {
      if (['ENOENT', 'ENOTDIR', 'EACCES'].includes(error.code)) continue
      throw error
    }
    const executable = realpathSync(file)
    ensure(executable === physical, 'Static compiler PATH selected a different Node executable')
    return { method: 'path', selected, physical, executable, path }
  }
  throw new Error('Static compiler PATH did not resolve node')
}
function runtimeArguments(action) {
  const invocation = action.command, argv = action.runtimeArgv ?? invocation, node = action.nodeResolution
  ensure(Array.isArray(invocation) && invocation.length && invocation.every(arg => typeof arg === 'string')
    && Array.isArray(argv) && argv.length === invocation.length && argv.every(arg => typeof arg === 'string')
    && isAbsolute(argv[0]) && equal(argv.slice(1), invocation.slice(1)), 'Static compiler raw invocation/runtime argv differs')
  if (action.runtimeArgv) {
    ensure(node && node.selected === argv[0] && isAbsolute(node.physical) && node.executable === node.physical,
      'Static compiler executable identity is incomplete')
    ensure(node.method === 'absolute' && isAbsolute(invocation[0]) && Object.keys(node).sort().join(',') === 'executable,method,physical,selected'
      || node.method === 'path' && invocation[0] === 'node' && typeof node.path === 'string' && !node.path.includes('\0')
        && Object.keys(node).sort().join(',') === 'executable,method,path,physical,selected',
    'Static compiler raw executable is not the reviewed Node invocation')
  } else ensure(isAbsolute(invocation[0]), 'Static compiler bare invocation has no executable identity witness')
  return argv
}
function companionExited(action) {
  if (action.exitCode === 0 && !action.signal) return true
  if (!(action.exitCode === null && action.signal === 'SIGTERM' || action.exitCode === 143 && !action.signal)) return false
  const requests = action.cleanupRequests
  return Array.isArray(requests) && !requests.some(item => item.success && item.signal !== 'SIGTERM') && requests.some(item => {
    const started = Date.parse(action.startedAt), requested = Date.parse(item.requestedAt)
    const completed = Date.parse(item.completedAt), ended = Date.parse(action.endedAt)
    return item.pid === action.pid && item.signal === 'SIGTERM' && item.success === true
      && ['process.kill', 'child.kill'].includes(item.via) && equal(item.observer, action.observer)
      && started <= requested && requested <= completed && completed <= ended && ended - requested <= COMPANION_CLEANUP_BOUND_MS
  })
}
function generatedSupport(before, after) {
  ensure(Array.isArray(before) && Array.isArray(after), 'Static source closure is missing')
  const original = new Map(before.map(item => [item.path, item])), current = new Map(after.map(item => [item.path, item]))
  ensure(original.size === before.length && current.size === after.length, 'Duplicate static source member')
  for (const [path, item] of original) ensure(equal(item, current.get(path)), `Static CLI mutated its original source/dependency closure: ${path}`)
  const added = after.filter(item => !original.has(item.path))
  ensure(added.every(item => GENERATED_SUPPORT_PATHS.has(item.path) && item.kind === 'file' && item.artifact), 'Static CLI added an unreviewed support file')
  return added
}
function companionPath(checkout, caseSourcePath) {
  ensure(typeof caseSourcePath === 'string' && caseSourcePath.startsWith('e2e/'), 'Companion case source is missing')
  safe(caseSourcePath)
  return join(checkout, '.tmp/e2e-ide-hmr-companion', caseSourcePath.slice(4).replaceAll('/', '__'))
}
function safe(path) {
  ensure(typeof path === 'string' && path && !isAbsolute(path) && !path.includes('\\') && !path.includes('\0')
    && path.split('/').every(part => part && part !== '.' && part !== '..'), 'Unsafe fixture artifact/member path')
  return path
}
function physicalNewPath(path) {
  let parent = resolve(path), parts = []
  while (!existsSync(parent)) { parts.unshift(basename(parent)); parent = dirname(parent) }
  return resolve(realpathSync(parent), ...parts)
}
function archive(root, directory) {
  root = realpathSync(root); directory = physicalNewPath(directory)
  ensure(inside(root, directory) && directory !== root, 'Fixture evidence directory escapes root')
  mkdirSync(directory, { recursive: true })
  return { root, directory, put: bytes => archiveCompilerBuildBytes({ root, directory, bytes }) }
}
function reader(root, loader, references = new Map()) {
  return (ref, label = 'fixture build') => {
    safe(ref?.path)
    const target = resolve(root, ref.path)
    if (!loader) ensure(inside(realpathSync(root), realpathSync(target)) && lstatSync(target).isFile() && !lstatSync(target).isSymbolicLink(), 'Fixture artifact escapes evidence root')
    const bytes = Buffer.from(loader ? loader(ref, label, false) : readFileSync(target))
    ensure(bytes.length === ref.bytes && hash(bytes) === ref.sha256, `Fixture artifact hash/size differs: ${ref.path}`)
    ensure(!references.has(ref.path) || equal(references.get(ref.path), ref), 'Conflicting fixture artifact references')
    references.set(ref.path, ref)
    return bytes
  }
}
function snapshot(directory, put, { exclude = new Set(), checkout, consumerGuard } = {}) {
  directory = realpathSync(directory)
  const files = []
  const walk = (base, prefix = '') => {
    for (const entry of readdirSync(base, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
      const path = prefix ? `${prefix}/${entry.name}` : entry.name
      if (exclude.has(path)) continue
      const file = join(base, entry.name)
      if (entry.isDirectory()) walk(file, path)
      else if (entry.isFile()) files.push({ path, kind: 'file', artifact: put(readFileSync(file)) })
      else {
        ensure(checkout && consumerGuard && entry.isSymbolicLink() && inside(checkout, realpathSync(file)), `Fixture snapshot contains an unbound symlink/special file: ${path}`)
        files.push({ path, kind: 'consumer-link', target: slash(relative(checkout, realpathSync(file))), consumerGuard })
      }
    }
  }
  walk(directory)
  return files
}
function readProject(projectPath) {
  projectPath = realpathSync(projectPath)
  const configPath = join(projectPath, 'project.config.json')
  ensure(lstatSync(configPath).isFile() && !lstatSync(configPath).isSymbolicLink(), 'Actual project config is not a regular file')
  const bytes = readFileSync(configPath), config = JSON.parse(bytes)
  const root = config.miniprogramRoot ?? '.'
  ensure(typeof root === 'string' && !isAbsolute(root) && !root.includes('\\') && !root.includes('\0'), 'Actual miniprogramRoot is unsafe')
  const outputRoot = resolve(projectPath, root)
  ensure(inside(projectPath, outputRoot) && outputRoot !== projectPath, 'Static profile requires a separate generated miniprogramRoot')
  return { projectPath, configPath, configBytes: bytes, outputRoot, outputRelative: slash(relative(projectPath, outputRoot)) }
}

/** Install in the actual case worker before its CLI build. The wrapper only
 * observes direct owned Node CLI `build <project> [--platform weapp]` invocations; unsupported
 * commands still run normally and cannot produce a formal checkpoint. */
export function installStaticFixtureBuildRecorder({ root, directory, compilerUseConfigPath, compilerBuild, expectedCommit,
  expectedProfile = COMPILER_BUILD_PROFILE, caseId, sourceSha256, caseSourcePath, fixtureCliPath }) {
  const a = archive(root, directory), load = reader(a.root)
  const checked = inspectCompilerBuildEvidence({ root: a.root, reference: compilerBuild, expectedCommit, expectedProfile })
  ensure(!checked.errors.length, checked.errors.join('\n'))
  const config = JSON.parse(readFileSync(compilerUseConfigPath))
  ensure(config.caseId === caseId && config.sourceSha256 === sourceSha256 && config.caseSourcePath === caseSourcePath, 'Fixture recorder differs from actual compiler consumer case')
  ensure(hash(readFileSync(join(config.evidenceRoot, config.membersReference.path))) === config.membersReference.sha256, 'Fixture recorder compiler guard changed')
  const cliRelativePath = expectedProfile === COMPILER_BUILD_PROFILE ? CLI : fixtureCliPath
  safe(cliRelativePath)
  const cliPath = realpathSync(join(config.checkout, cliRelativePath))
  ensure(inside(config.checkout, cliPath) && checked.attestation.source.files.some(item => item.path === cliRelativePath), 'Fixture CLI is not pinned owned compiler source')
  ensure(!recorderOwners.size, 'A static fixture build recorder already owns this worker')
  const ownerId = randomUUID(), original = childProcess.spawn, originalKill = process.kill, builds = [], pending = []
  const liveCompanions = new Map(), restoreChildKills = new Set()
  const controlledEnvironment = { NODE_OPTIONS: `--require=${JSON.stringify(join(dirname(compilerUseConfigPath), 'compiler-use-preload.cjs'))}`,
    NODE_PATH: '', WEAPP_CONTAINER_COMPILER_USE_CONFIG: compilerUseConfigPath }
  ensure(hash(readFileSync(join(dirname(compilerUseConfigPath), 'compiler-use-preload.cjs'))) === hash(readFileSync(new URL('./conformance-compiler-use-preload.cjs', import.meta.url))), 'Static compiler spawn has an unreviewed consumer preload')
  recorderOwners.add(ownerId)
  const record = { ...a, ownerId, config, compilerBuild, expectedCommit, expectedProfile, caseId, sourceSha256, caseSourcePath,
    cliRelativePath, cliPath, observer: { pid: process.pid, threadId }, builds, pending, disposed: false }
  function observeCleanup(candidate, via, signal, invoke) {
    const request = { pid: candidate.pid, signal: signal === undefined || signal === 15 ? 'SIGTERM' : signal,
      via, observer: record.observer, requestedAt: new Date().toISOString() }
    candidate.cleanupRequests.push(request)
    try { const result = invoke(); request.success = result === true; return result }
    catch (error) { request.success = false; request.errorCode = error.code ?? null; throw error }
    finally { request.completedAt = new Date().toISOString() }
  }
  function observedKill(pid, signal) {
    const owned = liveCompanions.get(pid)
    if (!owned || signal === 0 || owned.child.exitCode !== null || owned.child.signalCode !== null)
      return originalKill.apply(this, arguments)
    return observeCleanup(owned.candidate, 'process.kill', signal, () => originalKill.apply(this, arguments))
  }
  function observedSpawn(command, args = [], options = {}) {
    let candidate
    try {
      const cwd = realpathSync(options.cwd ?? process.cwd())
      const cliAt = args.findIndex(arg => typeof arg === 'string' && resolve(cwd, arg) === cliPath)
      if (cliAt < 0) return original.call(childProcess, command, args, options)
      candidate = { status: 'blocked', command: [command, ...args], cwd, startedAt: new Date().toISOString(), observer: record.observer,
        failure: 'Unsupported compiler CLI action: only the exact reviewed static target or owned companion is supported' }
      builds.push(candidate)
      const nodeResolution = selectedNode(command, options, cwd)
      ensure(cliAt === 0, candidate.failure)
      const staticBuild = staticBuildArguments(args, cliPath)
      const companion = args.length === 3 && args[1] === 'dev' && args[2] === '--non-interactive'
        && cwd === companionPath(config.checkout, caseSourcePath)
      ensure(staticBuild || companion, candidate.failure)
      const project = readProject(staticBuild ? resolve(cwd, args[2]) : cwd)
      if (staticBuild) {
        for (const action of builds) if (action.action === 'companion-dev' && action.status === 'closed' && !existsSync(action.projectPath)) {
          action.projectRemovedBeforeTargetAt = new Date().toISOString()
        }
      }
      ensure(!existsSync(project.outputRoot), 'Static compiler output existed before this build')
      const sourceBefore = snapshot(project.projectPath, a.put, { exclude: new Set(['.git', project.outputRelative]), checkout: config.checkout, consumerGuard: config.membersReference })
      const fixtureInputClosure = staticBuild && config.macroObservation
        ? a.put(JSON.stringify({ projectPath: project.projectPath, sourceBefore })) : undefined
      const controlledConsumer = staticBuild || Boolean(config.macroObservation)
      if (controlledConsumer) {
        for (const [key, value] of Object.entries(controlledEnvironment)) ensure(!options.env?.[key] || options.env[key] === value, `Static compiler spawn selected an unreviewed ${key}`)
        const environment = { ...(options.env ?? process.env), ...controlledEnvironment }
        // Both approved actions load the same observed compiler. The pinned
        // companion deliberately sanitizes NODE_OPTIONS, so restore its owned
        // loader too, while keeping target fixture provenance static-only.
        delete environment.WEAPP_CONTAINER_COMPILER_FIXTURE_INPUTS
        if (fixtureInputClosure) environment.WEAPP_CONTAINER_COMPILER_FIXTURE_INPUTS = JSON.stringify(fixtureInputClosure)
        options = { ...options, env: environment }
      } else ensure(!options.env?.NODE_OPTIONS || options.env.NODE_OPTIONS === controlledEnvironment.NODE_OPTIONS, 'Companion selected an unreviewed NODE_OPTIONS')
      Object.assign(candidate, { status: staticBuild ? 'building' : 'running', action: staticBuild ? 'static-build' : 'companion-dev',
        runtimeArgv: [process.execPath, ...args], nodeResolution,
        environment: controlledConsumer ? controlledEnvironment : { NODE_OPTIONS: options.env?.NODE_OPTIONS ?? '', NODE_PATH: options.env?.NODE_PATH ?? '' }, ...project,
        projectConfig: a.put(project.configBytes), outputAbsent: true, sourceBefore,
        ...(fixtureInputClosure ? { fixtureInputClosure } : {}), stdout: [], stderr: [] })
      candidate.startedAt = new Date().toISOString()
      delete candidate.failure
    } catch (error) {
      if (candidate) { candidate.status = 'blocked'; candidate.failure = error.message }
      else { candidate = { status: 'blocked', command: [command, ...args], failure: error.message }; builds.push(candidate) }
    }
    let child
    try { child = original.call(childProcess, command, args, options) }
    catch (error) { if (candidate) { candidate.status = 'blocked'; candidate.failure = error.message; candidate.endedAt = new Date().toISOString() } throw error }
    if (candidate) {
      candidate.pid = child.pid
      if (candidate.action === 'companion-dev' && candidate.status === 'running') {
        candidate.cleanupRequests = []
        liveCompanions.set(child.pid, { candidate, child })
        const originalChildKill = child.kill
        function observedChildKill(signal) {
          if (!liveCompanions.has(candidate.pid) || child.exitCode !== null || child.signalCode !== null || signal === 0)
            return originalChildKill.apply(this, arguments)
          return observeCleanup(candidate, 'child.kill', signal, () => originalChildKill.apply(this, arguments))
        }
        child.kill = observedChildKill
        const restore = () => { if (child.kill === observedChildKill) child.kill = originalChildKill; restoreChildKills.delete(restore) }
        restoreChildKills.add(restore)
        child.once('exit', () => { liveCompanions.delete(candidate.pid); restore() })
        child.once('error', () => { liveCompanions.delete(candidate.pid); restore() })
      }
      const captured = candidate.stdout && candidate.stderr
      if (captured) {
        child.stdout?.on('data', bytes => candidate.stdout.push(Buffer.from(bytes)))
        child.stderr?.on('data', bytes => candidate.stderr.push(Buffer.from(bytes)))
      }
      candidate.outputCaptured = !!child.stdout && !!child.stderr
      const settled = new Promise(accept => {
        child.once('error', error => { candidate.status = 'blocked'; candidate.failure = error.message; candidate.endedAt = new Date().toISOString(); accept() })
        child.once('close', (exitCode, signal) => {
          Object.assign(candidate, { exitCode, signal, endedAt: new Date().toISOString(),
            ...(captured ? { stdout: a.put(Buffer.concat(candidate.stdout)), stderr: a.put(Buffer.concat(candidate.stderr)) } : {}) })
          try {
            ensure(candidate.status !== 'blocked', candidate.failure)
            ensure(candidate.outputCaptured, 'CLI output streams were not captured')
            if (candidate.action === 'companion-dev') {
              ensure(companionExited(candidate), 'Companion did not exit normally or by observed bounded owned SIGTERM cleanup')
              candidate.sourceAfter = snapshot(candidate.projectPath, a.put, { exclude: new Set(['.git', candidate.outputRelative]), checkout: config.checkout, consumerGuard: config.membersReference })
              candidate.generated = snapshot(candidate.outputRoot, a.put)
              candidate.status = 'closed'
            } else {
              ensure(exitCode === 0 && !signal, 'Static CLI build failed')
              candidate.sourceAfter = snapshot(candidate.projectPath, a.put, { exclude: new Set(['.git', candidate.outputRelative]), checkout: config.checkout, consumerGuard: config.membersReference })
              candidate.generatedSupport = generatedSupport(candidate.sourceBefore, candidate.sourceAfter)
              candidate.generated = snapshot(candidate.outputRoot, a.put)
              ensure(candidate.generated.length && candidate.generated.some(item => item.path === 'app.json'), 'Static CLI generated no complete mini-program')
              ensure(equal(candidate.generated, snapshot(candidate.outputRoot, a.put)), 'Static CLI output changed while capturing')
              candidate.status = 'built'
            }
          } catch (error) { candidate.status = 'blocked'; candidate.failure = error.message }
          accept()
        })
      })
      // A companion watcher must already be closed before target compilation;
      // never wait indefinitely for a live unrelated watcher at checkpoint.
      if (candidate.action !== 'companion-dev') pending.push(settled)
    }
    return child
  }
  childProcess.spawn = observedSpawn; process.kill = observedKill; syncBuiltinESMExports()
  record.dispose = () => {
    if (record.disposed) return
    ensure(childProcess.spawn === observedSpawn, 'Static fixture recorder lost its scoped spawn ownership')
    ensure(process.kill === observedKill, 'Static fixture recorder lost its scoped signal ownership')
    childProcess.spawn = original; process.kill = originalKill
    for (const restore of restoreChildKills) restore()
    liveCompanions.clear()
    syncBuiltinESMExports(); recorderOwners.delete(ownerId); record.disposed = true
  }
  record.flush = () => flushStaticFixtureBuildRecorder({ recorder: record })
  return record
}

export async function captureOfficialFixtureCheckpoint({ recorder: r, runtimeObservation, checkpointId = 'static-initial' }) {
  await Promise.all(r.pending)
  const value = { schemaVersion: 1, kind: 'weapp-container.static-generated-checkpoint', profile: FIXTURE_BUILD_PROFILE, status: 'blocked',
    ownerId: r.ownerId, runId: r.config.runId, caseId: r.caseId, sourceSha256: r.sourceSha256, caseSourcePath: r.caseSourcePath,
    compilerBuild: r.compilerBuild, consumerGuard: r.config.membersReference, checkpointId,
    observer: r.observer, recorderDirectory: r.directory, runtimeObservation: r.put(JSON.stringify(runtimeObservation)), capturedAt: new Date().toISOString(), cliRelativePath: r.cliRelativePath }
  try {
    ensure(checkpointId === 'static-initial', 'Only static-initial checkpoint is supported; HMR generation is unproved')
    const actual = readProject(runtimeObservation.projectPath)
    const matches = r.builds.filter(build => build.status === 'built' && build.projectPath === actual.projectPath && build.outputRoot === actual.outputRoot)
    ensure(matches.length === 1 && r.builds.filter(build => build.projectPath === actual.projectPath).length === 1, 'Actual runtime project has no unique successful static CLI build; dev/watch/wrapper projects are unsupported')
    const build = matches[0]
    ensure(build.projectConfig.sha256 === hash(actual.configBytes), 'Opened project configuration differs from the compiled project')
    ensure(equal(build.generated, snapshot(actual.outputRoot, r.put)), 'Opened mini-program generation differs from the captured static CLI output')
    ensure(typeof runtimeObservation.installationId === 'string' && /^[0-9a-f]{64}$/.test(runtimeObservation.installationId)
      && Date.parse(runtimeObservation.recordedAt) >= Date.parse(build.endedAt) && Date.parse(runtimeObservation.recordedAt) <= Date.parse(value.capturedAt), 'Actual runtime installation/time observation is incomplete or preceded the build')
    const { configBytes, configPath, stdout, stderr, ...details } = build
    value.build = { ...details, stdout, stderr }; value.status = 'captured'
  } catch (error) { value.failure = error.message }
  const reference = r.put(JSON.stringify(value, null, 2) + '\n')
  const checkpointPath = join(r.directory, 'static-checkpoint.json')
  writeFileSync(checkpointPath, JSON.stringify(value, null, 2) + '\n')
  r.checkpoint = reference
  if (value.status !== 'captured') throw Object.assign(new Error(value.failure), { fixtureCheckpoint: reference })
  return { reference, attestation: value }
}

/** Flush after all case actions, before restoring the scoped observer. A
 * deleted cleanup project needs no live files; all observed later compiler
 * actions still make this static proof unusable. */
export async function flushStaticFixtureBuildRecorder({ recorder: r }) {
  await Promise.all(r.pending)
  const value = { schemaVersion: 1, kind: 'weapp-container.static-recorder-summary', profile: FIXTURE_BUILD_PROFILE,
    status: 'blocked', ownerId: r.ownerId, runId: r.config.runId, caseId: r.caseId, sourceSha256: r.sourceSha256,
    checkpoint: r.checkpoint ?? null, observedCliSpawns: r.builds.length, recordingEndedAt: new Date().toISOString(),
    actions: r.builds.map(({ stdout, stderr, configBytes, configPath, ...build }) => ({ ...build, ...(stdout && !Array.isArray(stdout) ? { stdout, stderr } : {}) })) }
  try {
    ensure(r.checkpoint, 'Static recorder has no runtime checkpoint')
    const checkpoint = JSON.parse(readFileSync(join(r.root, r.checkpoint.path)))
    const build = r.builds.find(build => build.pid === checkpoint.build?.pid)
    ensure(build && build.status === 'built', 'Static recorder observed no successful target build')
    for (const action of value.actions) {
      if (action.pid === build.pid) continue
      action.projectRemoved = !!action.projectPath && !existsSync(action.projectPath)
    }
    verifyActions(value.actions, checkpoint, r.config.checkout, r.caseSourcePath, r.config.membersReference, reader(r.root))
    if (existsSync(build.projectPath)) {
      const actual = readProject(build.projectPath)
      ensure(equal(build.generated, snapshot(actual.outputRoot, r.put)) && equal(build.sourceAfter,
        snapshot(build.projectPath, r.put, { exclude: new Set(['.git', build.outputRelative]), checkout: r.config.checkout, consumerGuard: r.config.membersReference })), 'Static source/generated generation changed after the runtime checkpoint')
      value.finalGuard = 'same-source-and-generated-closure'
    } else value.finalGuard = 'owned-project-cleaned-after-captured-generation'
    value.status = 'complete'
  } catch (error) { value.failure = error.message }
  const reference = r.put(JSON.stringify(value, null, 2) + '\n')
  writeFileSync(join(r.directory, 'static-recorder-summary.json'), JSON.stringify(value, null, 2) + '\n')
  return { reference, attestation: value }
}

export function reviewedFixturePackerClosure() {
  const files = PACKER_MODULES.map(path => ({ path, bytes: readFileSync(join(REPOSITORY, path)) }))
  const require = createRequire(join(REPOSITORY, PACKER)), dependencyRoot = dirname(require.resolve('acorn/package.json'))
  const walk = (directory, prefix = 'node_modules/acorn') => {
    for (const entry of readdirSync(directory, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
      const file = join(directory, entry.name), path = `${prefix}/${entry.name}`
      if (entry.isDirectory()) walk(file, path)
      else { ensure(entry.isFile(), 'Packer dependency closure contains a link/special file'); files.push({ path, bytes: readFileSync(file) }) }
    }
  }
  walk(dependencyRoot)
  return files.sort((a, b) => a.path.localeCompare(b.path))
}
function materialize(files, directory, load) {
  ensure(!existsSync(directory), 'Fixture packer input/executor directory already exists')
  mkdirSync(directory, { recursive: true })
  for (const item of files) {
    safe(item.path); ensure(item.kind === undefined || item.kind === 'file', 'Generated fixture closure cannot contain links')
    const file = join(directory, item.path); mkdirSync(dirname(file), { recursive: true }); writeFileSync(file, load(item.artifact), { flag: 'wx' })
  }
}
function resourceReferences(manifest, outputFiles) {
  const hashes = new Set()
  const visit = (value, key = '') => {
    if (Array.isArray(value)) value.forEach(child => visit(child))
    else if (value && typeof value === 'object') Object.entries(value).forEach(([key, child]) => visit(child, key))
    else if (typeof value === 'string') {
      if (['sha256', 'sourceSha256', 'executableSha256'].includes(key) && /^[0-9a-f]{64}$/.test(value)) hashes.add(value)
      if (['wxmlAst', 'wxssIr'].includes(key) && /^sha256:[0-9a-f]{64}$/.test(value)) hashes.add(value.slice(7))
    }
  }
  visit(manifest)
  const resources = {}
  for (const sha256 of [...hashes].sort()) {
    const file = outputFiles.find(item => item.path === `blobs/sha256/${sha256}`)
    ensure(file?.artifact.sha256 === sha256, 'Actual packer output blob closure is incomplete')
    resources[sha256] = file.artifact
  }
  return resources
}
export async function finalizeStaticFixtureBuild({ root, directory, checkpoint, compilerBuild, compilerUse, expectedCommit,
  expectedProfile = COMPILER_BUILD_PROFILE, timeoutMs = 30_000, fixtureId, recorderSummary }) {
  const a = archive(root, directory), load = reader(a.root)
  checkpoint = typeof checkpoint === 'string' ? a.put(readFileSync(checkpoint)) : checkpoint.reference ?? checkpoint
  const captured = JSON.parse(load(checkpoint))
  recorderSummary = recorderSummary?.reference ?? recorderSummary ?? join(captured.recorderDirectory, 'static-recorder-summary.json')
  if (typeof recorderSummary === 'string') recorderSummary = a.put(readFileSync(recorderSummary))
  const consumed = inspectCompilerUseEvidence({ root: a.root, reference: compilerUse, compilerBuild, expectedCommit, expectedProfile,
    caseId: captured.caseId, sourceSha256: captured.sourceSha256 })
  ensure(!consumed.errors.length, consumed.errors.join('\n'))
  verifyCheckpoint(captured, consumed.attestation, load, expectedProfile, consumed.validatedMembers)
  verifySummary(JSON.parse(load(recorderSummary)), captured, checkpoint, consumed.attestation, load, consumed.validatedMembers)
  const ownerId = randomUUID(), executor = join(a.directory, 'packer-executor'), input = join(a.directory, 'packer-input'), output = join(a.directory, 'packer-output')
  const modules = reviewedFixturePackerClosure().map(item => ({ path: item.path, artifact: a.put(item.bytes) }))
  materialize(modules, executor, load); materialize(captured.build.generated, input, load)
  const before = a.put(JSON.stringify(snapshot(input, a.put)))
  const preload = a.put(readFileSync(PRELOAD)), preloadPath = join(a.directory, 'packer-preload.mjs'), logPath = join(a.directory, 'packer-loads.jsonl')
  writeFileSync(preloadPath, load(preload), { flag: 'wx' }); writeFileSync(logPath, '', { flag: 'wx' })
  const configPath = join(a.directory, 'packer-config.json'), config = { log: logPath, files: modules.map(item => ({ file: join(executor, item.path), sha256: item.artifact.sha256, bytes: item.artifact.bytes })) }
  writeFileSync(configPath, JSON.stringify(config), { flag: 'wx' })
  const command = [process.execPath, '--import', preloadPath, join(executor, PACKER), input, '--out', output, '--compiler-commit', expectedCommit]
  const startedAt = new Date().toISOString(), stdout = [], stderr = []
  const child = spawn(command[0], command.slice(1), { cwd: executor, env: { PATH: process.env.PATH, NODE_OPTIONS: '', NODE_PATH: '', WEAPP_CONTAINER_FIXTURE_PACKER_CONFIG: configPath }, stdio: ['ignore', 'pipe', 'pipe'] })
  child.stdout.on('data', bytes => stdout.push(bytes)); child.stderr.on('data', bytes => stderr.push(bytes))
  let timedOut = false
  const timer = setTimeout(() => { timedOut = true; child.kill('SIGKILL') }, timeoutMs)
  const terminal = await new Promise(accept => { child.once('error', error => { stderr.push(Buffer.from(error.message)); accept({ exitCode: null, signal: 'spawn-error' }) }); child.once('close', (exitCode, signal) => accept({ exitCode, signal })) })
  clearTimeout(timer)
  const value = { schemaVersion: 1, kind: 'weapp-container.fixture-build', profile: FIXTURE_BUILD_PROFILE, status: 'blocked', ownerId,
    caseId: captured.caseId, sourceSha256: captured.sourceSha256, compilerBuild, compilerUse, checkpoint, recorderSummary,
    producer: a.put(readFileSync(MODULE)), packer: { modules, preload, executor, input, output, config: a.put(readFileSync(configPath)),
      command, run: { ...terminal, pid: child.pid ?? null, timedOut, startedAt, endedAt: new Date().toISOString(), stdout: a.put(Buffer.concat(stdout)), stderr: a.put(Buffer.concat(stderr)) },
      loads: a.put(readFileSync(logPath)) }, before }
  try {
    ensure(terminal.exitCode === 0 && !terminal.signal && !timedOut, 'Actual archived packer failed; inspect retained logs')
    value.after = a.put(JSON.stringify(snapshot(input, a.put)))
    ensure(equal(value.before, value.after), 'Actual packer mutated immutable generated input')
    ensure(equal(modules, snapshot(executor, a.put).map(({ path, artifact }) => ({ path, artifact }))), 'Packer executable module closure changed')
    value.outputFiles = snapshot(output, a.put)
    const manifest = JSON.parse(load(value.outputFiles.find(item => item.path === 'manifest.json')?.artifact))
    const fixtureValue = { id: fixtureId ?? `official-static-${captured.caseId}-${manifest.integrity.root}`, manifest }
    ensure(inspectExecutionFixture(fixtureValue, expectedCommit).status === 'passed', 'Actual packer did not produce a valid pinned fixture')
    value.fixture = a.put(JSON.stringify(fixtureValue, null, 2) + '\n'); value.resources = resourceReferences(manifest, value.outputFiles); value.status = 'built'
  } catch (error) { value.failure = error.message }
  const reference = a.put(JSON.stringify(value, null, 2) + '\n')
  return { reference, fixtureBuild: reference, attestation: value, fixture: value.fixture,
    fixtureValue: value.fixture ? JSON.parse(load(value.fixture)) : null, resources: value.resources }
}

function verifyActions(actions, checkpoint, checkout, caseSourcePath, consumerGuard, load, validatedMembers) {
  ensure(Array.isArray(actions) && actions.length, 'Static recorder has no compiler actions')
  const target = actions.filter(action => action.pid === checkpoint.build?.pid)
  ensure(target.length === 1 && equal(target[0], checkpoint.build), 'Static target action differs from checkpoint')
  const members = validatedMembers ?? JSON.parse(load(consumerGuard)).members
  const memberPaths = members.map(item => item.path)
  for (const action of actions) {
    if (action === target[0]) continue
    const expected = companionPath(checkout, caseSourcePath)
    ensure(action.action === 'companion-dev' && action.status === 'closed' && action.projectRemoved === true
      && action.projectPath === expected && action.cwd === expected && action.outputRoot === join(expected, action.outputRelative)
      && inside(expected, action.outputRoot) && action.outputRoot !== expected
      && !inside(expected, checkpoint.build.projectPath) && !inside(checkpoint.build.projectPath, expected)
      && equal(action.observer, checkpoint.observer) && action.command?.length === 4
      && runtimeArguments(action)[0] === runtimeArguments(checkpoint.build)[0] && action.command[1] === checkpoint.build.command[1]
      && action.command[2] === 'dev' && action.command[3] === '--non-interactive'
      && Number.isSafeInteger(action.pid) && action.pid > 0 && action.outputCaptured && action.outputAbsent
      && companionExited(action)
      && Date.parse(action.endedAt) >= Date.parse(action.startedAt)
      && Date.parse(action.projectRemovedBeforeTargetAt) >= Date.parse(action.endedAt)
      && Date.parse(action.projectRemovedBeforeTargetAt) <= Date.parse(checkpoint.build.startedAt)
      && Date.parse(action.endedAt) < Date.parse(checkpoint.build.startedAt),
    'Unrelated compiler action is not a closed, removed, disjoint owned companion before the target build')
    ensure(Array.isArray(action.sourceBefore) && action.sourceBefore.length && Array.isArray(action.sourceAfter) && action.sourceAfter.length
      && Array.isArray(action.generated) && action.generated.some(item => item.path === 'app.json'), 'Companion input/output closure is incomplete')
    for (const [files, links] of [[action.sourceBefore, true], [action.sourceAfter, true], [action.generated, false]]) {
      const names = new Set()
      for (const item of files) {
        safe(item.path); ensure(!names.has(item.path), 'Duplicate companion member'); names.add(item.path)
        if (item.kind === 'file') load(item.artifact)
        else ensure(links && item.kind === 'consumer-link' && equal(item.consumerGuard, consumerGuard)
          && memberPaths.some(path => path === item.target || path.startsWith(item.target + '/')), 'Companion has an unbound dependency or target-project link')
      }
    }
    const config = JSON.parse(load(action.projectConfig))
    ensure(action.sourceBefore.some(item => item.path === 'project.config.json' && equal(item.artifact, action.projectConfig))
      && resolve(expected, config.miniprogramRoot ?? '.') === action.outputRoot, 'Companion output is not its owned project root')
    load(action.stdout); load(action.stderr)
  }
}

// Live recording can validate action ownership and cleanup before compiler-use
// logs are sealed. Only the offline reader has the complete verified use; keep
// its actual startup proof separate so live flush never borrows partial logs.
function verifyObservedCompanionStartups(actions, checkpoint, use, load) {
  if (!use.macroObservation) return
  const logs = use.logs.flatMap(ref => load(ref).toString('utf8').trim().split('\n').map(line => JSON.parse(line)))
  for (const action of actions) {
    if (action.pid === checkpoint.build.pid) continue
    const startups = logs.filter(event => event.kind === 'start' && event.pid === action.pid && event.threadId === 0)
    ensure(equal(action.environment, checkpoint.build.environment) && startups.length === 1
      && equal(startups[0].argv, runtimeArguments(action)) && startups[0].cwd === action.cwd
      && startups[0].ppid === action.observer.pid && startups[0].nodeOptions === use.nodeOptions
      && !startups[0].fixtureInputClosure
      && logs.some(event => event.kind === 'load' && event.pid === action.pid
        && event.path?.startsWith('packages/weapp-vite/dist/') && event.loadedSourceSha256 === event.sha256),
    'Observed compiler companion lacks its exact controlled environment and actual compiler startup')
  }
}

function verifySummary(summary, checkpoint, checkpointReference, use, load, validatedMembers) {
  ensure(summary.schemaVersion === 1 && summary.kind === 'weapp-container.static-recorder-summary' && summary.profile === FIXTURE_BUILD_PROFILE
    && summary.status === 'complete' && summary.ownerId === checkpoint.ownerId && summary.runId === use.ownerId
    && summary.caseId === use.caseId && summary.sourceSha256 === use.sourceSha256
    && summary.checkpoint?.sha256 === checkpointReference.sha256 && summary.checkpoint?.bytes === checkpointReference.bytes
    && summary.observedCliSpawns === summary.actions?.length
    && ['same-source-and-generated-closure', 'owned-project-cleaned-after-captured-generation'].includes(summary.finalGuard)
    && Date.parse(summary.recordingEndedAt) >= Date.parse(checkpoint.capturedAt) && Date.parse(summary.recordingEndedAt) <= Date.parse(use.endedAt),
  'Static recorder summary is incomplete, changed generation, or observed later compiler actions')
  verifyActions(summary.actions, checkpoint, use.checkout, use.caseSourcePath, use.before, load, validatedMembers)
  verifyObservedCompanionStartups(summary.actions, checkpoint, use, load)
  for (const action of summary.actions) ensure(Date.parse(action.startedAt) >= Date.parse(use.startedAt)
    && Date.parse(action.endedAt) <= Date.parse(use.endedAt), 'Compiler action is outside the owned consumer interval')
  load(summary.checkpoint)
}

function verifyCheckpoint(c, use, load, expectedProfile, validatedMembers) {
  ensure(c.kind === 'weapp-container.static-generated-checkpoint' && c.schemaVersion === 1 && c.profile === FIXTURE_BUILD_PROFILE
    && c.status === 'captured' && c.checkpointId === 'static-initial', 'Static fixture checkpoint is absent, blocked, or a nonstatic generation')
  ensure(c.runId === use.ownerId && c.caseId === use.caseId && c.sourceSha256 === use.sourceSha256 && c.caseSourcePath === use.caseSourcePath
    && equal(c.compilerBuild, use.compilerBuild) && equal(c.consumerGuard, use.before), 'Fixture checkpoint belongs to another consumer/case/generation')
  const build = c.build, runtime = JSON.parse(load(c.runtimeObservation))
  ensure(runtime.status === undefined || runtime.status === 'passed', 'Fixture checkpoint captured a failed runtime observation')
  ensure(build.status === 'built' && build.exitCode === 0 && !build.signal && build.outputCaptured && build.outputAbsent === true && isAbsolute(build.projectPath)
    && isAbsolute(build.outputRoot) && inside(build.projectPath, build.outputRoot) && build.projectPath !== build.outputRoot, 'Static CLI build is failed, not fresh, or lacks an owned output')
  ensure(build.environment?.NODE_OPTIONS === use.nodeOptions && build.environment?.NODE_PATH === ''
    && typeof build.environment?.WEAPP_CONTAINER_COMPILER_USE_CONFIG === 'string', 'Static CLI did not inherit the exact controlled compiler consumer environment')
  ensure(runtime.projectPath === build.projectPath && /^[0-9a-f]{64}$/.test(runtime.installationId)
    && Date.parse(runtime.recordedAt) >= Date.parse(build.endedAt) && Date.parse(runtime.recordedAt) <= Date.parse(c.capturedAt)
    && Date.parse(build.startedAt) >= Date.parse(use.startedAt) && Date.parse(build.endedAt) >= Date.parse(build.startedAt)
    && Date.parse(c.capturedAt) <= Date.parse(use.endedAt), 'Fixture runtime project/installation/time is unrelated to compiler execution')
  ensure(equal(generatedSupport(build.sourceBefore, build.sourceAfter), build.generatedSupport ?? []) && build.sourceBefore.length
    && Array.isArray(build.generated) && build.generated.length, 'Fixture source or generated closure is missing or changed')
  const projectConfig = JSON.parse(load(build.projectConfig))
  ensure(build.sourceBefore.some(item => item.path === 'project.config.json' && equal(item.artifact, build.projectConfig)), 'Opened project config was not in the compiled source closure')
  ensure(resolve(build.projectPath, projectConfig.miniprogramRoot ?? '.') === build.outputRoot, 'Fixture generated root differs from actual opened project config')
  const expectedCli = expectedProfile === COMPILER_BUILD_PROFILE ? CLI : c.cliRelativePath
  ensure(c.cliRelativePath === expectedCli && staticBuildArguments(build.command?.slice(1), join(use.checkout, expectedCli))
    && resolve(build.cwd, build.command[3]) === build.projectPath, 'Fixture build did not execute the supported pinned static CLI/project')
  const runtimeArgv = runtimeArguments(build)
  // The consumer verifier already parsed, hashed and matched this exact
  // immutable member list. Reuse that internal output rather than rereading a
  // multi-megabyte shared guard for every case in the same audit.
  ensure(Array.isArray(validatedMembers) && Object.isFrozen(validatedMembers), 'Fixture consumer member closure was not internally verified')
  const consumerMembers = new Map(validatedMembers.map(item => [item.path, item]))
  const validateFiles = (files, links) => {
    const names = new Set()
    for (const item of files) {
      safe(item.path); ensure(!names.has(item.path), 'Duplicate fixture closure path'); names.add(item.path)
      if (item.kind === 'file') load(item.artifact)
      else ensure(links && item.kind === 'consumer-link' && equal(item.consumerGuard, use.before)
        && [...consumerMembers.keys()].some(path => path === item.target || path.startsWith(item.target + '/')), 'Fixture dependency link is not bound to actual consumer closure')
    }
  }
  validateFiles(build.sourceBefore, true); validateFiles(build.sourceAfter, true); validateFiles(build.generated, false); load(build.stdout); load(build.stderr)
  const logs = use.logs.flatMap(ref => load(ref).toString('utf8').trim().split('\n').map(line => JSON.parse(line)))
  const startup = logs.find(event => event.kind === 'start' && event.pid === build.pid)
  const consumerStartup = logs.find(event => event.kind === 'start' && event.pid === use.process.pid && event.threadId === 0)
  ensure(consumerStartup && runtimeArgv[0] === consumerStartup.argv?.[0], 'Fixture compiler Node differs from the actual root consumer executable')
  if (use.macroObservation) ensure(build.fixtureInputClosure && equal(startup?.fixtureInputClosure, build.fixtureInputClosure)
    && equal(JSON.parse(load(build.fixtureInputClosure)), { projectPath: build.projectPath, sourceBefore: build.sourceBefore }),
  'Fixture macro inputs differ from the actual unique static CLI action')
  ensure(startup && equal(startup.argv, runtimeArgv) && startup.ppid === c.observer.pid
    && logs.some(event => event.pid === build.pid && event.kind === 'load' && event.path?.startsWith('packages/weapp-vite/dist/')
      && event.loadedSourceSha256 === event.sha256 || expectedProfile !== COMPILER_BUILD_PROFILE && event.pid === build.pid && event.kind === 'load' && consumerMembers.get(event.path)?.origin === 'output'), 'Fixture CLI process was not the actual observed compiler consumer')
}

export function inspectConformanceFixtureBuildEvidence({ root, reference, compilerBuild, compilerUse, expectedCommit, caseId, sourceSha256,
  fixture, resources, runtimeObservation, expectedProfile = COMPILER_BUILD_PROFILE, loadArtifact, compilerBuildCache }) {
  const errors = [], references = new Map(), load = reader(root, loadArtifact, references)
  let attestation, compilerUseAttestation
  try {
    attestation = JSON.parse(load(reference))
    const a = attestation
    ensure(a.schemaVersion === 1 && a.kind === 'weapp-container.fixture-build' && a.profile === FIXTURE_BUILD_PROFILE && a.status === 'built', 'Fixture build is absent, blocked or unsupported')
    ensure(a.caseId === caseId && a.sourceSha256 === sourceSha256 && equal(a.compilerBuild, compilerBuild) && equal(a.compilerUse, compilerUse), 'Fixture build source/compiler consumer differs')
    const consumed = inspectCompilerUseEvidence({ root, reference: compilerUse, compilerBuild, expectedCommit, caseId, sourceSha256, expectedProfile, loadArtifact, compilerBuildCache })
    ensure(!consumed.errors.length, consumed.errors.join('\n'))
    compilerUseAttestation = consumed.attestation
    for (const ref of consumed.artifactReferences) references.set(ref.path, ref)
    const captured = JSON.parse(load(a.checkpoint)); verifyCheckpoint(captured, consumed.attestation, load, expectedProfile, consumed.validatedMembers)
    verifySummary(JSON.parse(load(a.recorderSummary)), captured, a.checkpoint, consumed.attestation, load, consumed.validatedMembers)
    if (runtimeObservation) ensure(equal(runtimeFacts(JSON.parse(load(captured.runtimeObservation))), runtimeFacts(runtimeObservation)), 'Fixture checkpoint belongs to a different official runtime observation')
    ensure(hash(load(a.producer)) === hash(readFileSync(MODULE)) && hash(load(a.packer.preload)) === hash(readFileSync(PRELOAD)), 'Fixture recorder/packer boundary differs from reviewed implementation')
    const actualModules = a.packer.modules
    const reviewed = reviewedFixturePackerClosure()
    ensure(equal(actualModules.map(item => ({ path: item.path, sha256: item.artifact.sha256, bytes: item.artifact.bytes })), reviewed.map(item => ({ path: item.path, sha256: hash(item.bytes), bytes: item.bytes.length }))), 'Archived packer module/dependency closure is not the reviewed implementation')
    actualModules.forEach(item => load(item.artifact))
    const p = a.packer, run = p.run
    ensure([p.executor, p.input, p.output].every(isAbsolute) && p.executor !== p.input && p.executor !== p.output && p.input !== p.output, 'Packer execution/input/output directories are not distinct absolute paths')
    ensure(run.exitCode === 0 && !run.signal && !run.timedOut && Number.isSafeInteger(run.pid) && run.pid > 0
      && Date.parse(run.endedAt) >= Date.parse(run.startedAt) && Date.parse(run.startedAt) >= Date.parse(captured.capturedAt), 'Actual packer invocation was not successful or preceded its input capture')
    ensure(equal(p.command, [p.command[0], '--import', join(dirname(p.executor), 'packer-preload.mjs'), join(p.executor, PACKER), p.input, '--out', p.output, '--compiler-commit', expectedCommit]), 'Packer invocation selected other inputs/compiler/options')
    load(run.stdout); load(run.stderr)
    ensure(equal(a.before, a.after) && equal(fileIdentities(JSON.parse(load(a.before))), fileIdentities(captured.build.generated)), 'Packer did not consume the captured complete generated closure unchanged')
    load(a.after)
    const moduleFiles = new Map(actualModules.map(item => [join(p.executor, item.path), item.artifact]))
    const config = JSON.parse(load(p.config))
    ensure(equal(config.files, actualModules.map(item => ({ file: join(p.executor, item.path), sha256: item.artifact.sha256, bytes: item.artifact.bytes }))), 'Packer loader config differs from archived modules')
    const events = load(p.loads).toString('utf8').trim().split('\n').map(line => JSON.parse(line))
    ensure(events.length && events.some(event => event.file === join(p.executor, PACKER)), 'Packer actual entry was not loaded')
    for (const event of events) ensure(event.pid === run.pid && event.sha256 === moduleFiles.get(event.file)?.sha256 && event.bytes === moduleFiles.get(event.file)?.bytes
      && (event.sourceSha256 === null || event.sourceSha256 === event.sha256)
      && Date.parse(event.at) >= Date.parse(run.startedAt) && Date.parse(event.at) <= Date.parse(run.endedAt), 'Packer loaded different modules or belongs to another process/time')
    const paths = new Set()
    for (const item of a.outputFiles) { safe(item.path); ensure(item.kind === 'file' && !paths.has(item.path), 'Invalid/duplicate packer output'); paths.add(item.path); load(item.artifact) }
    const fixtureValue = JSON.parse(load(a.fixture)), manifestOutput = a.outputFiles.find(item => item.path === 'manifest.json')
    ensure(inspectExecutionFixture(fixtureValue, expectedCommit).status === 'passed' && equal(JSON.parse(load(manifestOutput?.artifact)), fixtureValue.manifest), 'Recipe fixture differs from actual archived packer manifest')
    ensure(equal(a.resources, resourceReferences(fixtureValue.manifest, a.outputFiles)), 'Recipe resource closure differs from actual packer output')
    if (fixture) ensure(equal(fixtureValue, fixture.manifest ? fixture : JSON.parse(load(fixture))), 'Recipe substituted an unrelated hand-authored fixture')
    if (resources) {
      ensure(equal(Object.keys(a.resources).sort(), Object.keys(resources).sort()), 'Recipe substituted a different blob closure')
      for (const [sha256, ref] of Object.entries(resources)) ensure(ref.sha256 === a.resources[sha256].sha256 && ref.bytes === a.resources[sha256].bytes
        && load(ref).length === ref.bytes, 'Recipe substituted different blob resources')
    }
  } catch (error) { errors.push(error.message) }
  return { errors, attestation, compilerUseAttestation, artifactReferences: [...references.values()] }
}
